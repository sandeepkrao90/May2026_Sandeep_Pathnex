
# Setup Terraform on Amazon Linux EC2

# Launch EC2
*Amazon Linux 2023
*t2.micro
Attach IAM Role with:
*EC2FullAccess
*S3FullAccess
*DynamoDBFullAccess

#Connect to EC2.

ssh -i your-key.pem ec2-user@<EC2-PUBLIC-IP>

#Update and Install Terraform 

#update system software
sudo yum update -y        

 #install unzip and wget
sudo yum install -y unzip wget         

#Download Terraform setup

wget https://releases.hashicorp.com/terraform/1.8.5/terraform_1.8.5_linux_amd64.zip                  #d

#Unzip terraform download setup

unzip terraform_1.8.5_linux_amd64.zip

#Move terraform setup files to binaries
sudo mv terraform /usr/local/bin/

#Check Terraform version, will return version if installed
terraform version

#Create Project Directory

mkdir terraform-demo
cd terraform-demo

#Create Main File

vi main.tf

Add code to file (press i to insert)

provider "aws" {
  region = "ap-south-1"
}

resource "aws_instance" "demo_server" {
  ami           = "ami-0f58b397bc5c1f2e8"
  instance_type = "t2.micro"

  tags = {
    Name = "Terraform-Demo"
  }
}

Esc then :wq to save exit out of vi

#Now run Terraform commands to trigger Workflow

#Initialize working directory and download provider plugins and create terraform folder

terraform init

#Validates if code is written is OK or Not

terraform validate

#Terraform Plan shows whcih resources terraform will create, delete and modify

terraform plan

# Terraform apply creates resource

terraform apply

yes

Check resource created on AWS and then to delete use Terraform destroy

terraform destroy

Backend State management

#Create S3 Bucket
aws s3 mb s3://terraform-prod-state-demo

#Create DynamoDB Table
aws dynamodb create-table \
--table-name terraform-locks \
--attribute-definitions AttributeName=LockID,AttributeType=S \
--key-schema AttributeName=LockID,KeyType=HASH \
--billing-mode PAY_PER_REQUEST

#Backend configuration

terraform {
  backend "s3" {
    bucket         = "terraform-prod-state-demo"
    key            = "dev/terraform.tfstate"
    region         = "ap-south-1"
    dynamodb_table = "terraform-locks"
  }
}


#Reinitialize

terraform init

#Terraform migrates state to S3.
